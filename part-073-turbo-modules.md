# Part 073: Turbo Modules

## Turbo Modules คืออะไร?

Turbo Modules คือ evolution ของ Native Modules ใน React Native New Architecture ที่ออกแบบมาเพื่อแก้ไขปัญหาของ legacy bridge ซึ่งมีการ serialize/deserialize JSON ที่ช้า Turbo Modules ใช้ JSI (JavaScript Interface) ในการเรียก native code โดยตรงโดยไม่ผ่าน bridge

## ข้อดีของ Turbo Modules

1. **เร็วกว่า**: ไม่มีการ serialize/deserialize JSON
2. **Lazy Loading**: โหลด module เมื่อใช้งานจริงเท่านั้น
3. **Type Safety**: มี CodeGen สร้าง type definitions
4. **Synchronous calls**: เรียก native methods แบบ synchronous ได้

```objectivec
// RCTNativeCalendarModule.mm
#import "RCTNativeCalendarModule.h"
#import <React/RCTUtils.h>
#import <EventKit/EventKit.h>

@implementation RCTNativeCalendarModule
{
  EKEventStore *_eventStore;
}

- (instancetype)init {
  if (self = [super init]) {
    _eventStore = [[EKEventStore alloc] init];
  }
  return self;
}

// Turbo Module registration
- (std::shared_ptr<facebook::react::TurboModule>)getTurboModule:
    (const facebook::react::ObjCTurboModule::InitParams &)params
{
  return std::make_shared<facebook::react::NativeCalendarModuleSpecJSI>(params);
}

- (NSString *)getDeviceModel {
  return [[UIDevice currentDevice] model];
}

- (NSDictionary *)getConstants {
  return @{
    @"PERMISSION_WRITE": @"write",
    @"PERMISSION_READ": @"read",
    @"MAX_EVENTS": @(100)
  };
}

- (void)createCalendarEvent:(NSString *)title
                   location:(NSString *)location
                  startDate:(double)startDate
                    resolve:(RCTPromiseResolveBlock)resolve
                     reject:(RCTPromiseRejectBlock)reject
{
  [_eventStore requestAccessToEntityType:EKEntityTypeEvent
                               completion:^(BOOL granted, NSError *error) {
    if (!granted) {
      reject(@"PERMISSION_DENIED", @"Calendar access denied", error);
      return;
    }
    
    EKEvent *event = [EKEvent eventWithEventStore:self->_eventStore];
    event.title = title;
    event.location = location;
    event.startDate = [NSDate dateWithTimeIntervalSince1970:startDate / 1000.0];
    event.endDate = [event.startDate dateByAddingTimeInterval:3600];
    event.calendar = [self->_eventStore defaultCalendarForNewEvents];
    
    NSError *saveError = nil;
    if ([self->_eventStore saveEvent:event span:EKSpanThisEvent commit:YES error:&saveError]) {
      resolve(event.eventIdentifier);
    } else {
      reject(@"SAVE_FAILED", @"Failed to save event", saveError);
    }
  }];
}

- (void)getAllEvents:(RCTPromiseResolveBlock)resolve
             reject:(RCTPromiseRejectBlock)reject
{
  [_eventStore requestAccessToEntityType:EKEntityTypeEvent
                               completion:^(BOOL granted, NSError *error) {
    if (!granted) {
      reject(@"PERMISSION_DENIED", @"Calendar access denied", error);
      return;
    }
    
    NSDate *now = [NSDate date];
    NSDate *future = [now dateByAddingTimeInterval:30 * 24 * 3600];
    
    NSPredicate *predicate = [self->_eventStore predicateForEventsWithStartDate:now
                                                                        endDate:future
                                                                      calendars:nil];
    
    NSArray<EKEvent *> *events = [self->_eventStore eventsMatchingPredicate:predicate];
    
    NSMutableArray *result = [NSMutableArray array];
    for (EKEvent *event in events) {
      [result addObject:@{
        @"title": event.title ?: @"",
        @"location": event.location ?: @"",
        @"startDate": @([event.startDate timeIntervalSince1970] * 1000)
      }];
    }
    
    resolve(result);
  }];
}

- (void)deleteEvent:(NSString *)eventId
            resolve:(RCTPromiseResolveBlock)resolve
             reject:(RCTPromiseRejectBlock)reject
{
  EKEvent *event = [_eventStore eventWithIdentifier:eventId];
  if (!event) {
    reject(@"NOT_FOUND", @"Event not found", nil);
    return;
  }
  
  NSError *error = nil;
  BOOL success = [_eventStore removeEvent:event span:EKSpanThisEvent commit:YES error:&error];
  
  if (success) {
    resolve(@(YES));
  } else {
    reject(@"DELETE_FAILED", @"Failed to delete event", error);
  }
}

@end
```

## 3. Android Implementation

### 3.1 Java/Kotlin Implementation

```kotlin
// NativeCalendarModuleImpl.kt
package com.reactnativecalendar

import android.Manifest
import android.content.ContentValues
import android.content.pm.PackageManager
import android.net.Uri
import android.provider.CalendarContract
import androidx.core.content.ContextCompat
import com.facebook.react.bridge.Promise
import com.facebook.react.bridge.ReactApplicationContext
import com.facebook.react.bridge.WritableArray
import com.facebook.react.bridge.WritableNativeArray
import com.facebook.react.bridge.WritableNativeMap
import com.reactnativecalendar.NativeCalendarModuleSpec
import java.util.TimeZone

class NativeCalendarModuleImpl(
  private val reactContext: ReactApplicationContext
) : NativeCalendarModuleSpec(reactContext) {
  
  override fun getName() = NAME
  
  override fun getTypedExportedConstants(): Map<String, Any> {
    return mapOf(
      "PERMISSION_WRITE" to Manifest.permission.WRITE_CALENDAR,
      "PERMISSION_READ" to Manifest.permission.READ_CALENDAR,
      "MAX_EVENTS" to 100
    )
  }
  
  override fun getDeviceModel(): String {
    return android.os.Build.MODEL
  }
  
  override fun createCalendarEvent(
    title: String,
    location: String,
    startDate: Double,
    promise: Promise
  ) {
    if (ContextCompat.checkSelfPermission(
      reactContext,
      Manifest.permission.WRITE_CALENDAR
    ) != PackageManager.PERMISSION_GRANTED) {
      promise.reject("PERMISSION_DENIED", "Calendar write permission not granted")
      return
    }
    
    try {
      val values = ContentValues().apply {
        put(CalendarContract.Events.TITLE, title)
        put(CalendarContract.Events.EVENT_LOCATION, location)
        put(CalendarContract.Events.DTSTART, startDate.toLong())
        put(CalendarContract.Events.DTEND, startDate.toLong() + 3600000)
        put(CalendarContract.Events.EVENT_TIMEZONE, TimeZone.getDefault().id)
        put(CalendarContract.Events.CALENDAR_ID, 1)
      }
      
      val uri: Uri? = reactContext.contentResolver.insert(
        CalendarContract.Events.CONTENT_URI,
        values
      )
      
      if (uri != null) {
        promise.resolve(uri.lastPathSegment)
      } else {
        promise.reject("SAVE_FAILED", "Failed to create event")
      }
    } catch (e: Exception) {
      promise.reject("ERROR", e.message, e)
    }
  }
  
  override fun getAllEvents(promise: Promise) {
    try {
      val projection = arrayOf(
        CalendarContract.Events._ID,
        CalendarContract.Events.TITLE,
        CalendarContract.Events.EVENT_LOCATION,
        CalendarContract.Events.DTSTART
      )
      
      val results: WritableArray = WritableNativeArray()
      
      reactContext.contentResolver.query(
        CalendarContract.Events.CONTENT_URI,
        projection,
        null, null,
        "${CalendarContract.Events.DTSTART} ASC"
      )?.use { cursor ->
        while (cursor.moveToNext()) {
          val event = WritableNativeMap().apply {
            putString("title", cursor.getString(1) ?: "")
            putString("location", cursor.getString(2) ?: "")
            putDouble("startDate", cursor.getLong(3).toDouble())
          }
          results.pushMap(event)
        }
      }
      
      promise.resolve(results)
    } catch (e: Exception) {
      promise.reject("ERROR", e.message, e)
    }
  }
  
  override fun deleteEvent(eventId: String, promise: Promise) {
    try {
      val uri = Uri.withAppendedPath(
        CalendarContract.Events.CONTENT_URI,
        eventId
      )
      val deleted = reactContext.contentResolver.delete(uri, null, null)
      promise.resolve(deleted > 0)
    } catch (e: Exception) {
      promise.reject("ERROR", e.message, e)
    }
  }
  
  companion object {
    const val NAME = "CalendarModule"
  }
}
```

```typescript
    throw new Error('createCalendarEvent not available');
  },

    if (!module) return [];
    return module.getAllEvents?.() ?? [];
  }
```

### Step 2: iOS Swift Implementation

```swift
// NativeFileSystemModule.swift
import Foundation

@objc(FileSystemModule)
class NativeFileSystemModule: NSObject {
  
  @objc static func requiresMainQueueSetup() -> Bool { return false }
  
  @objc func getConstants() -> [String: Any] {
    return [
      "DocumentDirectory": NSSearchPathForDirectoriesInDomains(.documentDirectory, .userDomainMask, true).first ?? "",
      "CachesDirectory": NSSearchPathForDirectoriesInDomains(.cachesDirectory, .userDomainMask, true).first ?? "",
      "TempDirectory": NSTemporaryDirectory()
    ]
  }
  
  @objc func readFile(_ path: String, resolve: @escaping RCTPromiseResolveBlock, reject: @escaping RCTPromiseRejectBlock) {
    do {
      let content = try String(contentsOfFile: path, encoding: .utf8)
      resolve(content)
    } catch {
      reject("READ_ERROR", "Cannot read file: \(error.localizedDescription)", error)
    }
  }
  
  @objc func writeFile(_ path: String, content: String, resolve: @escaping RCTPromiseResolveBlock, reject: @escaping RCTPromiseRejectBlock) {
    do {
      try content.write(toFile: path, atomically: true, encoding: .utf8)
      resolve(true)
    } catch {
      reject("WRITE_ERROR", "Cannot write file: \(error.localizedDescription)", error)
    }
  }
  
  @objc func deleteFile(_ path: String, resolve: @escaping RCTPromiseResolveBlock, reject: @escaping RCTPromiseRejectBlock) {
    do {
      try FileManager.default.removeItem(atPath: path)
      resolve(true)
    } catch {
      reject("DELETE_ERROR", "Cannot delete file: \(error.localizedDescription)", error)
    }
  }
  
  @objc func exists(_ path: String, resolve: RCTPromiseResolveBlock, reject: RCTPromiseRejectBlock) {
    resolve(FileManager.default.fileExists(atPath: path))
  }
  
  @objc func listDirectory(_ path: String, resolve: @escaping RCTPromiseResolveBlock, reject: @escaping RCTPromiseRejectBlock) {
    do {
      let contents = try FileManager.default.contentsOfDirectory(atPath: path)
      let fileInfos = contents.map { name -> [String: Any] in
        let fullPath = (path as NSString).appendingPathComponent(name)
        let attrs = try? FileManager.default.attributesOfItem(atPath: fullPath)
        
        return [
          "name": name,
          "path": fullPath,
          "size": (attrs?[.size] as? Int) ?? 0,
          "isDirectory": (attrs?[.type] as? FileAttributeType) == .typeDirectory,
          "lastModified": ((attrs?[.modificationDate] as? Date)?.timeIntervalSince1970 ?? 0) * 1000
        ]
      }
      resolve(fileInfos)
    } catch {
      reject("LIST_ERROR", "Cannot list directory: \(error.localizedDescription)", error)
    }
  }
  
  @objc func createDirectory(_ path: String, resolve: @escaping RCTPromiseResolveBlock, reject: @escaping RCTPromiseRejectBlock) {
    do {
      try FileManager.default.createDirectory(
        atPath: path,
        withIntermediateDirectories: true,
        attributes: nil
      )
      resolve(true)
    } catch {
      reject("CREATE_DIR_ERROR", error.localizedDescription, error)
    }
  }
  
  @objc func moveFile(_ from: String, to: String, resolve: @escaping RCTPromiseResolveBlock, reject: @escaping RCTPromiseRejectBlock) {
    do {
      try FileManager.default.moveItem(atPath: from, toPath: to)
      resolve(true)
    } catch {
      reject("MOVE_ERROR", error.localizedDescription, error)
    }
  }
  
  @objc func copyFile(_ from: String, to: String, resolve: @escaping RCTPromiseResolveBlock, reject: @escaping RCTPromiseRejectBlock) {
    do {
      try FileManager.default.copyItem(atPath: from, toPath: to)
      resolve(true)
    } catch {
      reject("COPY_ERROR", error.localizedDescription, error)
    }
  }
  
  @objc func getFileInfo(_ path: String, resolve: @escaping RCTPromiseResolveBlock, reject: @escaping RCTPromiseRejectBlock) {
    guard let attrs = try? FileManager.default.attributesOfItem(atPath: path) else {
      reject("INFO_ERROR", "Cannot get file info", nil)
      return
    }
    
    resolve([
      "name": (path as NSString).lastPathComponent,
      "path": path,
      "size": (attrs[.size] as? Int) ?? 0,
      "isDirectory": (attrs[.type] as? FileAttributeType) == .typeDirectory,
      "lastModified": ((attrs[.modificationDate] as? Date)?.timeIntervalSince1970 ?? 0) * 1000
    ])
  }
}
```

### Step 3: JavaScript Usage

```typescript
// FileSystemService.ts
import FileSystemModule from './NativeFileSystem';

export class FileSystem {
  static readonly DocumentDirectory = FileSystemModule.getConstants().DocumentDirectory;
  static readonly CachesDirectory = FileSystemModule.getConstants().CachesDirectory;
  static readonly TempDirectory = FileSystemModule.getConstants().TempDirectory;
  
  static async readJSON<T>(path: string): Promise<T> {
    const content = await FileSystemModule.readFile(path);
    return JSON.parse(content) as T;
  }
  
  static async writeJSON(path: string, data: any): Promise<boolean> {
    const content = JSON.stringify(data, null, 2);
    return FileSystemModule.writeFile(path, content);
  }
  
  static async appendToFile(path: string, content: string): Promise<boolean> {
    const exists = await FileSystemModule.exists(path);
    if (exists) {
      const existing = await FileSystemModule.readFile(path);
      return FileSystemModule.writeFile(path, existing + content);
    }
    return FileSystemModule.writeFile(path, content);
  }
  
  static async ensureDirectory(dirPath: string): Promise<void> {
    const exists = await FileSystemModule.exists(dirPath);
    if (!exists) {
      await FileSystemModule.createDirectory(dirPath);
    }
  }
  
  static async readLines(path: string): Promise<string[]> {
    const content = await FileSystemModule.readFile(path);
    return content.split('\n').filter(line => line.length > 0);
  }
}
```

```tsx
// FileManagerComponent.tsx
import React, { useState, useEffect } from 'react';
import { View, Text, FlatList, TouchableOpacity, StyleSheet } from 'react-native';
import { FileSystem } from './FileSystemService';

const FileManagerComponent: React.FC = () => {
  const [files, setFiles] = useState<any[]>([]);
  const [currentPath, setCurrentPath] = useState(FileSystem.DocumentDirectory);
  
  useEffect(() => {
    loadFiles();
  }, [currentPath]);
  
  const loadFiles = async () => {
    try {
      const fileList = await require('./NativeFileSystem').default.listDirectory(currentPath);
      setFiles(fileList);
    } catch (error) {
      console.error('Error loading files:', error);
    }
  };
  
  const formatSize = (bytes: number): string => {
    if (bytes < 1024) return `${bytes} B`;
    if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
    return `${(bytes / (1024 * 1024)).toFixed(1)} MB`;
  };
  
  const renderFile = ({ item }: { item: any }) => (
    <TouchableOpacity
      style={styles.fileItem}
      onPress={() => {
        if (item.isDirectory) {
          setCurrentPath(item.path);
        }
      }}
    >
      <Text style={styles.fileName}>
        {item.isDirectory ? '📁' : '📄'} {item.name}
      </Text>
      {!item.isDirectory && (
        <Text style={styles.fileSize}>{formatSize(item.size)}</Text>
      )}
    </TouchableOpacity>
  );
  
  return (
    <View style={styles.container}>
      <Text style={styles.path} numberOfLines={1}>{currentPath}</Text>
      <FlatList
        data={files}
        renderItem={renderFile}
        keyExtractor={item => item.path}
      />
    </View>
  );
};

const styles = StyleSheet.create({
  container: { flex: 1 },
  path: { padding: 10, backgroundColor: '#eee', fontSize: 12 },
  fileItem: {
    flexDirection: 'row',
    justifyContent: 'space-between',
    padding: 15,
    borderBottomWidth: 1,
    borderBottomColor: '#eee'
  },
  fileName: { fontSize: 16 },
  fileSize: { fontSize: 14, color: '#666' }
});

export default FileManagerComponent;
```

## Tips สำหรับ Turbo Modules

1. **ใช้ TypeScript Spec เสมอ** - CodeGen จะสร้าง native code ที่ถูกต้อง
2. **Lazy Loading** - Turbo Modules โหลดเมื่อใช้งาน ประหยัด startup time
3. **Type Safety** - TypeScript types จะถูก enforce ทั้งฝั่ง JS และ Native
4. **Testing** - ทดสอบ spec ก่อน implement native code

## สรุป

Turbo Modules เป็นก้าวสำคัญสำหรับ React Native ที่ช่วย:
- ลด overhead จาก bridge serialization
- เพิ่ม type safety ด้วย CodeGen
- ปรับปรุง startup time ด้วย lazy loading
- รองรับ synchronous native calls
